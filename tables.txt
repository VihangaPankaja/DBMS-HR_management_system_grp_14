CREATE TABLE Annual_leave (
    Annual_ID VARCHAR(20),
    Leave_ID VARCHAR(20),
    num_dates INT,
    start_date date,
    end_date date,
    status VARCHAR(20),
    primary key (Annual_ID),
    foreign key (Leave_ID) references Leave(Leave_ID)
);

CREATE TABLE casual_leave (
    casual_ID VARCHAR(20),
    Leave_ID VARCHAR(20),
    num_dates INT,
    start_date date,
    end_date date,
    status VARCHAR(20),
    primary key (casual_ID),
    foreign key (Leave_ID) references Leave(Leave_ID)
);

CREATE TABLE Nopay_leave (
    Nopay_ID VARCHAR(20),
    Leave_ID VARCHAR(20),
    num_dates INT,
    start_date date,
    end_date date,
    status VARCHAR(20),
    primary key (Nopay_ID),
    foreign key (Leave_ID) references Leave(Leave_ID)
);

CREATE TABLE Maternity_leave (
    Maternity_ID VARCHAR(20),
    Leave_ID VARCHAR(20),
    num_dates INT,
    start_date date,
    end_date date,
    status VARCHAR(20),
    primary key (casual_ID),
    foreign key (Leave_ID) references Leave(Leave_ID)
);

CREATE TABLE Leave (
    Leave_ID VARCHAR(20),
    ID VARCHAR(20),
    status VARCHAR(20),
    Typeleave_ID VARCHAR(20),
    primary key (Leave_ID),
    foreign key (ID) references Employee(ID)
    foreign key (Typeleave_ID) references Typeleave(Typeleave_ID)
);

CREATE TABLE Typeleave(
    Typeeave_ID VARCHAR(20),
    Leave_ID VARCHAR(20),
    max_dates INT,
    primary key (Typeeave_ID)
);



create table Employee (
    ID varchar(20),
    payroll_ID varchar(20),
    Emergency_ID varchar(20),
    organization_ID varchar(20),
    attribute_ID varchar(20),
    detail_ID varchar(20),
    Leave_ID varchar(20),
    name varchar(20),
    birth_day date,
    matrial_status varchar(20)
    primary key (ID),
    foreign key (payroll_ID) references Payroll(payroll_ID),
    foreign key (Emergency_ID) references Emergency_contact(Emergency_ID),
    foreign key (oraganization_ID) references Organization(oraganization_ID),
    foreign key (attribute_ID) references attributes(attribute_ID),
    foreign key (detail_ID) references Job_detail(detail_ID),
    foreign key (Leave_ID) references Leave(Leave_ID)
);
     
create table Payroll (
    payroll_ID varchar(20),
    amount int,
    paygrade int,
    primary key (payroll_ID)
);
    
create table Emergency_contact(
    Emergency_ID varchar(20),
    ID varchar(20),
    name varchar(20),
    relationship varchar(20),
    contact_number int,
    primary key (Emergency_ID),
    foreign key (ID) references Employee(ID)
);

create table Organization(
    organization_ID varchar(20),
    name varchar(20),
    address varchar(20),
    reistration_namber varchar(20),
    primary key (organization_ID)
);
  
create table attributes(
    attribute_ID varchar(20),
    nationality varchar(20),
    religious varchar(20),
    primary key (attribute_ID)
);
  
create table Job_detail(
    deatail_ID varchar(20),
    job_title varchar(20),
    employment_status varchar(20),
    contract varchar(20),
    paygrade int,
    job_status varchar(20),
    primary key (detail_ID)
